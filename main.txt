import streamlit as st
from openai import OpenAI
from dotenv import load_dotenv
import os
import fitz  # PyMuPDF
import pandas as pd
from io import BytesIO
import json

# Carrega variáveis de ambiente
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
print("Chave da API da OpenAI carregada com sucesso.", api_key)  # Para depuração, remova em produção

    st.error("Erro: A chave da API da OpenAI não foi encontrada. Certifique-se de que OPENAI_API_KEY está definida no seu arquivo .env.")

client = OpenAI(api_key=api_key)


@st.cache_data
def extrair_texto_pdf(uploaded_file):
    try:
        pdf_bytes = uploaded_file.read()
        doc = fitz.open(stream=pdf_bytes, filetype="pdf")
        texto = ""
        for pagina in doc:
            texto += pagina.get_text()
        return texto
    except Exception as e:
        st.error(f"Erro ao extrair texto do PDF: {e}")
        return None


def gerar_excel_documentos(dados_completos):   
    """Gera um arquivo Excel com múltiplas abas a partir de dados completos."""
    if not dados_completos:
        return None
    output = BytesIO()
    try:
        with pd.ExcelWriter(output, engine="openpyxl") as writer:
            # Pasta de trabalho 1: Resumo do Edital (Ficha Técnica)
            df_resumo = pd.DataFrame([dados_completos.get("Resumo do Edital", {})])
            # Reordenar colunas para que as importantes fiquem no início, se desejar
            colunas_resumo = [
                "Objeto", "Modalidade do Pregão", "Tipo de Julgamento",
                "Data e Horário de Abertura", "Plataforma do Pregão",
                "Valor Estimado Total", "Endereço de Entrega", "Critérios de Avaliação",
                "Validade dos Documentos Solicitados", "Período de Envio dos Documentos",
                "Prazo de Envio da Proposta Readequada", "Exige Catálogo e Período de Apresentação",
                "Exigência de Atestado do Objeto ou Quantitativo e Sua Porcentagem"
            ]
            # Adiciona colunas que podem não estar na lista_documentos_geral_e_especifica, mas existem no DataFrame
            for col in df_resumo.columns:
                if col not in colunas_resumo:
                    colunas_resumo.append(col)
            df_resumo = df_resumo[colunas_resumo]
            df_resumo.to_excel(writer, sheet_name="Resumo do Edital", index=False)

            # Pasta de trabalho 2: Documentos de Habilitação
            df_documentacao = pd.DataFrame(dados_completos.get("Documentos de Habilitacao", []))
            df_documentacao.to_excel(writer, sheet_name="Documentos de Habilitacao", index=False)
            
            # Pasta de trabalho 3: Documentos de Credenciamento
            df_credenciamento = pd.DataFrame(dados_completos.get("Documentos de Credenciamento", []))
            df_credenciamento.to_excel(writer, sheet_name="Documentos de Credenciamento", index=False)

            # Pasta de trabalho 4: Garantias
            df_garantias = pd.DataFrame(dados_completos.get("Garantias Exigidas", []))
            df_garantias.to_excel(writer, sheet_name="Garantias", index=False)

        output.seek(0)
        return output
    except Exception as e:
        st.error(f"Erro ao gerar o arquivo Excel: {e}")
        return None

            2.  **Preencha TODOS os campos** no objeto "Resumo do Edital". Se a informação não for encontrada, indique explicitamente "Não informado" ou deixe em branco, mas não omita o campo.
            3.  Para as listas de documentos (Habilitação, Credenciamento) e Garantias, inclua **TODOS os itens encontrados no edital**, mesmo que pareçam óbvios ou genéricos.
